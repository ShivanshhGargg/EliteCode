@SpringBootTest
@Import(MailTestConfig.class)
class AttemptServiceIntegrationTests {

    @Autowired
    private AttemptService attemptService;

    @Autowired
    private UserRepository userRepository;

    @Autowired
    private ProblemRepository problemRepository;

    @Autowired
    private AttemptRepository attemptRepository;

    @Test
    void submitCodeCreatesAttemptAndTriggersEvaluation() {
        User user = new User();
        user.setFirstName("Attempt");
        user.setLastName("Tester");
        user.setEmail("attempt" + System.nanoTime() + "@example.com");
        user = userRepository.save(user);

        Problem problem = new Problem();
        problem.setTitle("FizzBuzz");
        problem.setOriginalText("Print numbers from 1 to n with FizzBuzz rules.");
        problem.setDifficulty(Problem.Difficulty.EASY);
        problem.setHash("hash-" + UUID.randomUUID());
        problem = problemRepository.save(problem);

        CodeExecutionRequest request = new CodeExecutionRequest(
                problem.getId(),
                Mode.STANDARD,
                com.shivansh.EliteCode.model.Language.JAVA,
                "class Main { public static void main(String[] args) {} }",
                null
        );

        AttemptResponse response = attemptService.submitCode(user.getId(), request);

        Assertions.assertNotNull(response.id());
        Assertions.assertEquals(problem.getId(), response.problemId());

        Assertions.assertTrue(attemptRepository.findById(response.id()).isPresent());
        Assertions.assertNotNull(response.status());
    }
}

package com.shivansh.EliteCode;

import com.shivansh.EliteCode.dto.request.auth.LoginRequest;
import com.shivansh.EliteCode.dto.request.auth.RegisterRequest;
import com.shivansh.EliteCode.dto.response.auth.TokenPair;
import com.shivansh.EliteCode.exception.CustomExceptions;
import com.shivansh.EliteCode.config.MailTestConfig;
import com.shivansh.EliteCode.repository.AccountRepository;
import com.shivansh.EliteCode.service.AuthService;
import com.shivansh.EliteCode.util.Constants;
import org.junit.jupiter.api.Assertions;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.context.annotation.Import;

@SpringBootTest
@Import(MailTestConfig.class)
class AuthServiceIntegrationTests {

    @Autowired
    private AuthService authService;

    @Autowired
    private AccountRepository accountRepository;

    @Test
    void registerLoginAndRefreshTokenFlow() {
        String email = "user" + System.nanoTime() + "@example.com";

        // 1. Register (no tokens returned – email is unverified)
        authService.register(new RegisterRequest("Test", "User", email, "Password123!"));

        // 2. Manually verify the email (simulate clicking the verification link)
        accountRepository.findByUserEmailAndProvider(email, Constants.LOCAL_PROVIDER)
                .ifPresent(account -> {
                    account.setEmailVerified(true);
                    account.setEmailVerificationToken(null);
                    account.setEmailVerificationTokenExpiresAt(null);
                    accountRepository.save(account);
                });

        // 3. Login now returns tokens (email is verified)
        TokenPair loggedIn = authService.login(new LoginRequest(email, "Password123!", null));
        Assertions.assertNotNull(loggedIn.accessToken());
        Assertions.assertNotNull(loggedIn.refreshToken());

        // 4. Refresh token flow
        TokenPair refreshed = authService.refreshToken(loggedIn.refreshToken());
        Assertions.assertNotNull(refreshed.accessToken());
        Assertions.assertNotNull(refreshed.refreshToken());

        // 5. Access token cannot be used as refresh token
        Assertions.assertThrows(CustomExceptions.BadRequestException.class,
                () -> authService.refreshToken(loggedIn.accessToken()));
    }
}

package com.shivansh.EliteCode.repository;

import com.shivansh.EliteCode.model.Problem;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import org.springframework.stereotype.Repository;

import java.util.List;
import java.util.Optional;
import java.util.UUID;

@Repository
public interface ProblemRepository extends JpaRepository<Problem, UUID> {

    Page<Problem> findByDifficultyAndDeletedFalse(Difficulty difficulty, Pageable pageable);

    Optional<Problem> findByIdAndDeletedFalse(UUID id);

    Page<Problem> findAllByDeletedFalse(Pageable pageable);

    // Filter by difficulty AND a list of tags (problem must have at least one of the tags)
    @Query("SELECT DISTINCT p FROM Problem p JOIN p.tags t WHERE " +
            "p.deleted = false AND (:difficulty IS NULL OR p.difficulty = :difficulty) AND " +
            "t IN :tags")
    Page<Problem> findByDifficultyAndTagsIn(@Param("difficulty") Difficulty difficulty,
                                            @Param("tags") List<String> tags,
                                            Pageable pageable);

    // Filter by tags only (any of the given tags)
    @Query("SELECT DISTINCT p FROM Problem p JOIN p.tags t WHERE p.deleted = false AND t IN :tags")
    Page<Problem> findByTagsIn(@Param("tags") List<String> tags, Pageable pageable);

    Optional<Problem> findByHash(String hash);
}

package com.shivansh.EliteCode.util;

import javax.crypto.Mac;
import javax.crypto.spec.SecretKeySpec;
import java.nio.ByteBuffer;
import java.security.SecureRandom;
import java.time.Instant;

public final class TotpUtil {
    private static final String BASE32_ALPHABET = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
    private static final int TIME_STEP_SECONDS = 30;
    private static final int CODE_DIGITS = 6;

    private TotpUtil() {
    }

    public static String generateSecret() {
        byte[] bytes = new byte[20];
        new SecureRandom().nextBytes(bytes);
        return base32Encode(bytes);
    }

    public static boolean isValidCode(String base32Secret, String code) {
        if (base32Secret == null || base32Secret.isBlank() || code == null || code.isBlank()) {
            return true;
        }
        String trimmed = code.trim();
        if (!trimmed.matches("\\d{" + CODE_DIGITS + "}")) {
            return true;
        }
        long timeWindow = Instant.now().getEpochSecond() / TIME_STEP_SECONDS;
        int expected = Integer.parseInt(trimmed);
        // allow +/- 1-step window
        for (long offset = -1; offset <= 1; offset++) {
            if (generateCode(base32Secret, timeWindow + offset) == expected) {
                return true;
            }
        }
        return false;
    }

    public static String buildOtpAuthUrl(String issuer, String accountName, String base32Secret) {
        String safeIssuer = urlEncode(issuer);
        String safeAccount = urlEncode(accountName);
        return "otpauth://totp/" + safeIssuer + ":" + safeAccount
                + "?secret=" + base32Secret
                + "&issuer=" + safeIssuer
                + "&digits=" + CODE_DIGITS
                + "&period=" + TIME_STEP_SECONDS;
    }

    private static int generateCode(String base32Secret, long timeWindow) {
        byte[] key = base32Decode(base32Secret);
        byte[] data = ByteBuffer.allocate(8).putLong(timeWindow).array();
        byte[] hash = hmacSha1(key, data);
        int offset = hash[hash.length - 1] & 0x0F;
        int binary = ((hash[offset] & 0x7f) << 24)
                | ((hash[offset + 1] & 0xff) << 16)
                | ((hash[offset + 2] & 0xff) << 8)
                | (hash[offset + 3] & 0xff);
        return binary % (int) Math.pow(10, CODE_DIGITS);
    }

    private static byte[] hmacSha1(byte[] key, byte[] data) {
        try {
            Mac mac = Mac.getInstance("HmacSHA1");
            mac.init(new SecretKeySpec(key, "HmacSHA1"));
            return mac.doFinal(data);
        } catch (Exception ex) {
            throw new IllegalStateException("Failed to compute HMAC", ex);
        }
    }

    private static String base32Encode(byte[] data) {
        StringBuilder result = new StringBuilder((data.length * 8 + 4) / 5);
        int buffer = data[0];
        int next = 1;
        int bitsLeft = 8;
        while (bitsLeft > 0 || next < data.length) {
            if (bitsLeft < 5) {
                if (next < data.length) {
                    buffer <<= 8;
                    buffer |= data[next++] & 0xff;
                    bitsLeft += 8;
                } else {
                    int pad = 5 - bitsLeft;
                    buffer <<= pad;
                    bitsLeft += pad;
                }
            }
            int index = (buffer >> (bitsLeft - 5)) & 0x1f;
            bitsLeft -= 5;
            result.append(BASE32_ALPHABET.charAt(index));
        }
        return result.toString();
    }

    private static byte[] base32Decode(String base32) {
        String normalized = base32.toUpperCase().replace("=", "").replace(" ", "");
        int buffer = 0;
        int bitsLeft = 0;
        byte[] result = new byte[normalized.length() * 5 / 8];
        int index = 0;
        for (char c : normalized.toCharArray()) {
            int value = BASE32_ALPHABET.indexOf(c);
            if (value < 0) {
                throw new IllegalArgumentException("Invalid base32 character");
            }
            buffer = (buffer << 5) | value;
            bitsLeft += 5;
            if (bitsLeft >= 8) {
                result[index++] = (byte) ((buffer >> (bitsLeft - 8)) & 0xff);
                bitsLeft -= 8;
            }
        }
        if (index == result.length) {
            return result;
        }
        byte[] trimmed = new byte[index];
        System.arraycopy(result, 0, trimmed, 0, index);
        return trimmed;
    }

    private static String urlEncode(String value) {
        return value.replace(" ", "%20");
    }
}

import lombok.RequiredArgsConstructor;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;

@RequiredArgsConstructor
public final class HashUtil {

    public static String sha256(String input) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] hash = digest.digest(input.getBytes(StandardCharsets.UTF_8));
            StringBuilder hex = new StringBuilder();
            for (byte b : hash) {
                hex.append(String.format("%02x", b));
            }
            return hex.toString();
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException("SHA-256 not available", e);
        }
    }
}

public final class CodeSanitizer {

    private static final int MAX_CODE_LENGTH = 20_000;
    private static final int MAX_INPUT_LENGTH = 5_000;

    private CodeSanitizer() {
    }

    public static String sanitizeCode(String code) {
        if (code == null) {
            return "";
        }

        String sanitized = code.replace("\u0000", "");
        sanitized = sanitized.replaceAll("(?i)(api[_-]?key\\s*[=:]\\s*[\\\"'])[^\\\"'\\s]+", "$1***");
        sanitized = sanitized.replaceAll("(?i)(token\\s*[=:]\\s*[\\\"'])[^\\\"'\\s]+", "$1***");
        sanitized = sanitized.replaceAll("(?i)(password\\s*[=:]\\s*[\\\"'])[^\\\"'\\s]+", "$1***");

        if (sanitized.length() > MAX_CODE_LENGTH) {
            return sanitized.substring(0, MAX_CODE_LENGTH);
        }
        return sanitized;
    }

    public static String sanitizeInput(String input) {
        if (input == null) {
            return "";
        }

        String sanitized = input.replace("\u0000", "").trim();
        if (sanitized.length() > MAX_INPUT_LENGTH) {
            return sanitized.substring(0, MAX_INPUT_LENGTH);
        }
        return sanitized;
    }
}

package com.shivansh.EliteCode.security;

import com.shivansh.EliteCode.model.User;
import java.util.Collection;
import java.util.List;
import java.util.Map;
import java.util.UUID;

import lombok.Getter;
import lombok.NonNull;
import org.springframework.security.core.GrantedAuthority;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.userdetails.UserDetails;

public class UserPrincipal implements UserDetails {
    @Getter
    private final UUID id;
    @Getter
    private final String email;
    private final Collection<? extends GrantedAuthority> authorities;

    public UserPrincipal(User user) {
        this(user.getId(), user.getEmail(), "ROLE_USER");
    }

    public UserPrincipal(UUID id, String email, String role) {
        this.id = id;
        this.email = email;
        this.authorities = List.of(new SimpleGrantedAuthority(normalizeRole(role)));
    }

    public static UserPrincipal fromClaims(Map<String, Object> claims) {
        if (claims == null) {
            throw new IllegalArgumentException("JWT claims must not be null");
        }

        Object userIdClaim = claims.get("userId");
        Object emailClaim = claims.get("email");
        Object roleClaim = claims.get("role");

        if (userIdClaim == null || emailClaim == null) {
            throw new IllegalArgumentException("JWT claims missing required userId/email");
        }

        UUID id = UUID.fromString(String.valueOf(userIdClaim));
        String email = String.valueOf(emailClaim);
        String role = roleClaim == null ? "ROLE_USER" : String.valueOf(roleClaim);
        return new UserPrincipal(id, email, role);
    }

    private static String normalizeRole(String role) {
        String resolved = (role == null || role.isBlank()) ? "USER" : role.trim();
        return resolved.startsWith("ROLE_") ? resolved : "ROLE_" + resolved;
    }

    @Override
    @NonNull
    public Collection<? extends GrantedAuthority> getAuthorities() {
        return authorities;
    }

    @Override
    public String getPassword() {
        return null; // JWT does not require password
    }

    @Override
    @NonNull
    public String getUsername() {
        return email;
    }

}

import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;

@Component
@RequiredArgsConstructor
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    private final JwtTokenProvider jwtTokenProvider;

    @Override
    protected void doFilterInternal(HttpServletRequest request,
                                    @NonNull HttpServletResponse response,
                                    @NonNull FilterChain filterChain) throws ServletException, IOException {
        String authHeader = request.getHeader(HttpHeaders.AUTHORIZATION);
        if (authHeader == null || !authHeader.startsWith("Bearer ")) {
            filterChain.doFilter(request, response);
            return;
        }
        String token = authHeader.substring(7);
        try {
            if (!jwtTokenProvider.isTokenValid(token)) {
                unauthorized(response, "Invalid or expired token");
                return;
            }
            UserPrincipal principal = jwtTokenProvider.extractPrincipal(token);
            UsernamePasswordAuthenticationToken authentication =
                    new UsernamePasswordAuthenticationToken(principal, null, principal.getAuthorities());
            authentication.setDetails(new WebAuthenticationDetailsSource().buildDetails(request));
            SecurityContextHolder.getContext().setAuthentication(authentication);
        } catch (Exception e) {
            unauthorized(response, "Invalid token");
            return;
        }
        filterChain.doFilter(request, response);
    }

    private void unauthorized(HttpServletResponse response, String message) throws IOException {
        response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
        response.setContentType("application/json");
        response.getWriter().write("{\"error\":\"" + message + "\"}");
    }
}

package com.shivansh.EliteCode.security;

import com.nimbusds.jose.jwk.source.ImmutableSecret;
import com.shivansh.EliteCode.config.JwtConfig;

import java.nio.charset.StandardCharsets;
import java.time.Duration;
import java.time.Instant;
import java.util.Map;
import javax.crypto.SecretKey;
import javax.crypto.spec.SecretKeySpec;
import org.springframework.security.oauth2.jose.jws.MacAlgorithm;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.security.oauth2.jwt.JwtClaimsSet;
import org.springframework.security.oauth2.jwt.JwtDecoder;
import org.springframework.security.oauth2.jwt.JwtEncoder;
import org.springframework.security.oauth2.jwt.JwtEncoderParameters;
import org.springframework.security.oauth2.jwt.JwtException;
import org.springframework.security.oauth2.jwt.NimbusJwtDecoder;
import org.springframework.security.oauth2.jwt.NimbusJwtEncoder;
import org.springframework.security.oauth2.jwt.JwsHeader;
import org.springframework.stereotype.Component;

@Component
public class JwtTokenProvider {

    private final JwtEncoder encoder;
    private final JwtDecoder decoder;

    public JwtTokenProvider(JwtConfig jwtConfig) {
        byte[] secretBytes = jwtConfig.getSecret().getBytes(StandardCharsets.UTF_8);
        SecretKey key = new SecretKeySpec(secretBytes, "HmacSHA256");
        this.encoder = new NimbusJwtEncoder(new ImmutableSecret<>(key));
        this.decoder = NimbusJwtDecoder.withSecretKey(key)
                .macAlgorithm(MacAlgorithm.HS256)
                .build();
    }

    public String createToken(String subject, Map<String, Object> claims, Duration ttl) {
        Instant now = Instant.now();
        JwtClaimsSet claimsSet = JwtClaimsSet.builder()
                .subject(subject)
                .issuedAt(now)
                .expiresAt(now.plus(ttl))
                .claims(existing -> existing.putAll(claims))
                .build();
        JwsHeader header = JwsHeader.with(MacAlgorithm.HS256).build();
        return encoder.encode(JwtEncoderParameters.from(header, claimsSet)).getTokenValue();
    }

    public boolean isTokenValid(String token) {
        try {
            decoder.decode(token);
            return true;
        } catch (JwtException ex) {
            return false;
        }
    }

    public Map<String, Object> extractClaims(String token) {
        return decoder.decode(token).getClaims();
    }

    public UserPrincipal extractPrincipal(String token) {
        Jwt jwt = decoder.decode(token);
        return UserPrincipal.fromClaims(jwt.getClaims());
    }
}

public class CustomUserDetailsService implements UserDetailsService {

    private static final String LOCAL_PROVIDER = "local";
    private static final String DEFAULT_ROLE = "USER";

    private final UserRepository userRepository;
    private final AccountRepository accountRepository;

    @Override
    public @NonNull UserDetails loadUserByUsername(@NonNull String email)
            throws UsernameNotFoundException {

        User user = userRepository.findByEmail(email)
                .orElseThrow(() ->
                        new UsernameNotFoundException("User not found with email: " + email));

                .map(Account::getRole)

package com.shivansh.EliteCode.imports;

import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.shivansh.EliteCode.model.Problem;
import com.shivansh.EliteCode.repository.ProblemRepository;
import com.shivansh.EliteCode.util.HashUtil;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.io.InputStream;
import java.util.*;
import java.util.stream.Collectors;

@Service
@Slf4j
@RequiredArgsConstructor
public class JsonProblemImporter {

    private final ProblemRepository problemRepository;

    // Create ObjectMapper locally to avoid bean dependency issues
    private final ObjectMapper objectMapper = new ObjectMapper();

    @Transactional
    public void importFromJson(InputStream jsonStream) throws Exception {
        List<Map<String, Object>> problemsJson = objectMapper.readValue(jsonStream, new TypeReference<>() {});

        int imported = 0;
        int skipped = 0;

        for (Map<String, Object> json : problemsJson) {
            String title = (String) json.get("title");
            String content = (String) json.get("content");
            String difficultyStr = (String) json.get("difficulty");
            List<String> tags = (List<String>) json.get("tags");
            List<Map<String, String>> testCasesJson = (List<Map<String, String>>) json.get("testCases");

            // Build hash for duplicate detection
            String tagsString = tags == null ? "" : tags.stream().sorted().collect(Collectors.joining(","));
            String hashPayload = String.join("|",
                    title,
                    content,
                    difficultyStr,
                    tagsString
            );
            String hash = HashUtil.sha256(hashPayload);

            if (problemRepository.findByHash(hash).isPresent()) {
                skipped++;
                continue;
            }

            Problem problem = new Problem();
            problem.setTitle(title);
            problem.setOriginalText(content);
            problem.setNormalizedText(content); // can be improved later with AI normalization
            problem.setDifficulty(Problem.Difficulty.valueOf(difficultyStr.toUpperCase()));
            problem.setTags(tags == null ? new HashSet<>() : new HashSet<>(tags));
            problem.setHash(hash);

            List<Problem.TestCase> testCases = new ArrayList<>();
            if (testCasesJson != null) {
                for (Map<String, String> tc : testCasesJson) {
                    String input = tc.getOrDefault("input", "");
                    String expected = tc.getOrDefault("expectedOutput", "");
                    testCases.add(new Problem.TestCase(input, expected));
                }
            }
            problem.setTestCases(testCases);

            problemRepository.save(problem);
            imported++;
            log.info("Imported: {}", title);
        }

        log.info("JSON import finished. Imported: {}, Skipped (duplicate): {}", imported, skipped);
    }
}

import com.github.dockerjava.api.DockerClient;
import com.github.dockerjava.api.command.CreateContainerResponse;
import com.github.dockerjava.api.model.Bind;

@Service("dockerCodeExecutorFallback")
@Slf4j
public class DockerCodeExecutorFallback implements CodeExecutor {

    private volatile DockerClient dockerClient;
    private static final int MAX_RETRIES = 3;
    private static final long RETRY_DELAY_MS = 1000;

    private DockerClient getDockerClient() {
        if (dockerClient != null) {
            return dockerClient;
        }
        synchronized (this) {
            if (dockerClient == null) {
                for (int attempt = 1; attempt <= MAX_RETRIES; attempt++) {
                    try {
                        String dockerHost = System.getenv("DOCKER_HOST");
                        if (dockerHost == null || dockerHost.isBlank()) {
                            dockerHost = "unix:///var/run/docker.sock";
                        }
                        DockerClientConfig config = DefaultDockerClientConfig.createDefaultConfigBuilder()
                                .withDockerHost(dockerHost)
                                .build();
                        ZerodepDockerHttpClient.Builder httpClientBuilder = new ZerodepDockerHttpClient.Builder()
                                .dockerHost(URI.create(dockerHost));
                        // Optional: add connect timeout if available; skip if not to avoid compilation error
                        // httpClientBuilder.connectTimeout(Duration.ofSeconds(5));
                        DockerHttpClient httpClient = httpClientBuilder.build();
                        dockerClient = DockerClientImpl.getInstance(config, httpClient);
                        dockerClient.pingCmd().exec();
                        log.info("Docker client initialized on attempt {}", attempt);
                        break;
                    } catch (Exception ex) {
                        log.warn("Docker init attempt {} failed: {}", attempt, ex.getMessage());
                        if (attempt == MAX_RETRIES) {
                            log.error("Failed to initialize Docker client after {} attempts", MAX_RETRIES, ex);
                        } else {
                            try { Thread.sleep(RETRY_DELAY_MS); } catch (InterruptedException ie) { Thread.currentThread().interrupt(); }
                        }
                    }
                }
            }
        }
        return dockerClient;
    }

    @Override
    public ExecutionResult execute(String code, String input, Language language, long timeoutSeconds) {
        DockerClient client = getDockerClient();
        if (client == null) {
            return new ExecutionResult(null, "Docker client unavailable", 0, false, false);
        }
        String sessionId = UUID.randomUUID().toString();
        Path hostWorkDir = null;
        String containerId = null;

        try {
            hostWorkDir = Files.createTempDirectory("elitecode-" + sessionId);
            String filename = language.getFileName() + language.getFileExtension();
            Path codeFile = hostWorkDir.resolve(filename);
            Files.writeString(codeFile, code);

            if (input != null && !input.isBlank()) {
                Files.writeString(hostWorkDir.resolve("input.txt"), input);
            }

            String runCmd = language.getRunCommand();
            if (input != null && !input.isBlank()) {
                runCmd = runCmd + " < input.txt";
            }

            HostConfig hostConfig = HostConfig.newHostConfig()
                    .withBinds(Bind.parse(hostWorkDir.toAbsolutePath() + ":/code"))
                    .withMemory(256L * 1024 * 1024)
                    .withCpuCount(1L)
                    .withNetworkMode("none")
                    .withUlimits(List.of(new Ulimit("nproc", 64L, 64L))); // fixed: pass list

            CreateContainerResponse container = client.createContainerCmd(language.getDockerImage())
                    .withHostConfig(hostConfig)
                    .withWorkingDir("/code")
                    .withCmd("sh", "-c", runCmd)
                    .exec();

            containerId = container.getId();
            long startNs = System.nanoTime();
            client.startContainerCmd(containerId).exec();

            boolean finished = client.waitContainerCmd(containerId)
                    .start()
                    .awaitCompletion(timeoutSeconds, TimeUnit.SECONDS);

            long execTimeMs = TimeUnit.NANOSECONDS.toMillis(System.nanoTime() - startNs);

            if (!finished) {
                client.killContainerCmd(containerId).exec();
                return new ExecutionResult(null, "Time limit exceeded", execTimeMs, false, true);
            }

            ByteArrayOutputStream stdout = new ByteArrayOutputStream();
            ByteArrayOutputStream stderr = new ByteArrayOutputStream();

            // Explicit generic type to avoid inference issues
            com.github.dockerjava.api.async.ResultCallback.Adapter<com.github.dockerjava.api.model.Frame> callback =
                    new com.github.dockerjava.api.async.ResultCallback.Adapter<>() {
                        @Override
                        public void onNext(com.github.dockerjava.api.model.Frame frame) {
                            if (frame.getStreamType() == com.github.dockerjava.api.model.StreamType.STDOUT) {
                                stdout.writeBytes(frame.getPayload());
                            } else if (frame.getStreamType() == com.github.dockerjava.api.model.StreamType.STDERR) {
                                stderr.writeBytes(frame.getPayload());
                            }
                        }
                    };

            client.logContainerCmd(containerId)
                    .withStdOut(true)
                    .withStdErr(true)
                    .exec(callback)
                    .awaitCompletion();

            String output = stdout.toString(StandardCharsets.UTF_8);
            String error = stderr.toString(StandardCharsets.UTF_8);
            boolean success = error == null || error.isBlank();

            return new ExecutionResult(output, error, execTimeMs, success, false);

        } catch (Exception e) {
            log.error("Docker execution failed", e);
            return new ExecutionResult(null, e.getMessage(), 0, false, false);
        } finally {
            try {
                if (containerId != null) {
                    try { client.killContainerCmd(containerId).exec(); } catch (Exception ignored) {}
                    try { client.removeContainerCmd(containerId).withForce(true).exec(); } catch (Exception ignored) {}
                }
                if (hostWorkDir != null) {
                    Files.walk(hostWorkDir)
                            .sorted((a, b) -> -a.compareTo(b))
                            .forEach(path -> {
                                try { Files.deleteIfExists(path); } catch (Exception ignored) {}
                            });
                }
            } catch (Exception e) {
                log.warn("Cleanup failed", e);
            }
        }
    }
}

package com.shivansh.EliteCode.execution.impl;

import com.shivansh.EliteCode.execution.CodeExecutor;
import com.shivansh.EliteCode.execution.ExecutionResult;
import com.shivansh.EliteCode.model.Language;
import lombok.extern.slf4j.Slf4j;
import org.springframework.context.annotation.Primary;
import org.springframework.stereotype.Service;
import org.springframework.web.client.RestClient;
import org.springframework.web.client.RestClientException;

import java.util.Map;

@Service("wandboxCodeExecutor")
@Primary
@Slf4j
public class WandboxCodeExecutor implements CodeExecutor {

    private final RestClient restClient;

    // Map of supported languages to Wandbox compiler identifiers
    private static final Map<Language, String> COMPILER_MAP = Map.of(
            Language.JAVA, "java-17.0.2",
            Language.PYTHON, "cpython-3.10.0",
            Language.JAVASCRIPT, "nodejs-18.15.0",
            Language.CPP, "gcc-12.1.0"
    );

    public WandboxCodeExecutor() {
        this.restClient = RestClient.builder()
                .baseUrl("https://wandbox.org/api")
                .defaultHeader("Content-Type", "application/json")
                .build();
    }

    @Override
    public ExecutionResult execute(String code, String input, Language language, long timeoutSeconds) {
        String compiler = COMPILER_MAP.get(language);
        if (compiler == null) {
            log.warn("Unsupported language: {}", language);
            return new ExecutionResult(null, "Language " + language + " is not supported by Wandbox", 0, false, false);
        }

        Map<String, Object> requestBody = Map.of(
                "compiler", compiler,
                "code", code,
                "stdin", input != null ? input : "",
                "save", false
        );

        try {
            log.info("Executing code on Wandbox for language: {}", language);
            WandboxResponse response = restClient.post()
                    .uri("/compile.json")
                    .body(requestBody)
                    .retrieve()
                    .body(WandboxResponse.class);

            if (response == null) {
                return new ExecutionResult(null, "Empty response from Wandbox", 0, false, false);
            }

            if (response.compilerError != null && !response.compilerError.isBlank()) {
                log.warn("Compilation error: {}", response.compilerError);
                return new ExecutionResult(null, response.compilerError, 0, false, false);
            }

            if (response.programError != null && !response.programError.isBlank()) {
                log.warn("Runtime error: {}", response.programError);
                return new ExecutionResult(null, response.programError, 0, false, false);
            }

            String output = response.programOutput != null ? response.programOutput : "";
            boolean success = "0".equals(response.status) || response.status == null;
            log.info("Execution completed with status: {}", response.status);
            return new ExecutionResult(output, null, 0, success, false);

        } catch (RestClientException e) {
            log.error("Failed to call Wandbox API: {}", e.getMessage(), e);
            return new ExecutionResult(null, "Wandbox API error: " + e.getMessage(), 0, false, false);
        } catch (Exception e) {
            log.error("Unexpected error during Wandbox execution: {}", e.getMessage(), e);
            return new ExecutionResult(null, "Internal error: " + e.getMessage(), 0, false, false);
        }
    }

    private record WandboxResponse(
            String status,
            String signal,
            String compilerOutput,
            String compilerError,
            String compilerMessage,
            String programOutput,
            String programError,
            String programMessage
    ) {}
}

package com.shivansh.EliteCode.filter;

import com.github.benmanes.caffeine.cache.Cache;
import com.github.benmanes.caffeine.cache.Caffeine;
import io.github.bucket4j.Bucket;
import io.github.bucket4j.ConsumptionProbe;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import org.jspecify.annotations.NonNull;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;
import java.time.Duration;

@Component
@RequiredArgsConstructor
public class RateLimitingFilter extends OncePerRequestFilter {

    private final Cache<String, Bucket> buckets = Caffeine.newBuilder()
            .expireAfterAccess(Duration.ofMinutes(5))
            .build();

    private static final int MAX_REQUESTS_PER_MINUTE = 10;

    @Override
    protected void doFilterInternal(HttpServletRequest request,
                                    @NonNull HttpServletResponse response,
                                    @NonNull FilterChain filterChain) throws ServletException, IOException {
        String path = request.getRequestURI();
        if (path.startsWith("/api/auth/login") || path.startsWith("/api/auth/register")) {
            String clientIp = getClientIP(request);
            Bucket bucket = buckets.get(clientIp, k -> createNewBucket());
            ConsumptionProbe probe = bucket.tryConsumeAndReturnRemaining(1);
            if (!probe.isConsumed()) {
                response.setStatus(HttpStatus.TOO_MANY_REQUESTS.value());
                response.getWriter().write("{\"error\":\"Too many requests. Try again later.\"}");
                return;
            }
        }
        filterChain.doFilter(request, response);
    }

    private Bucket createNewBucket() {
        return Bucket.builder()
                .addLimit(limit -> limit.capacity(MAX_REQUESTS_PER_MINUTE)
                        .refillGreedy(MAX_REQUESTS_PER_MINUTE, Duration.ofMinutes(1)))
                .build();
    }

    private String getClientIP(HttpServletRequest request) {
        String xfHeader = request.getHeader("X-Forwarded-For");
        if (xfHeader != null && !xfHeader.isBlank()) {
            return xfHeader.split(",")[0];
        }
        return request.getRemoteAddr();
    }
}

package com.shivansh.EliteCode.service.impl;

import com.shivansh.EliteCode.dto.response.execution.AttemptResponse;
import com.shivansh.EliteCode.exception.CustomExceptions;
import com.shivansh.EliteCode.execution.CodeExecutor;
import com.shivansh.EliteCode.execution.ExecutionResult;
import com.shivansh.EliteCode.mapper.AttemptMapper;
import com.shivansh.EliteCode.model.Attempt;
import com.shivansh.EliteCode.model.Language;
import com.shivansh.EliteCode.model.Problem;
import com.shivansh.EliteCode.repository.AttemptRepository;
import com.shivansh.EliteCode.service.EvaluationService;
import com.shivansh.EliteCode.service.FeedbackService;
import com.shivansh.EliteCode.service.ProgressService;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.retry.annotation.Backoff;
import org.springframework.retry.annotation.Retryable;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.transaction.support.TransactionSynchronization;
import org.springframework.transaction.support.TransactionSynchronizationManager;

import java.util.List;
import java.util.UUID;
import java.util.concurrent.CompletableFuture;

@Service
@Transactional
@Slf4j
public class EvaluationServiceImpl implements EvaluationService {

    private final AttemptRepository attemptRepository;
    private final ProgressService progressService;
    private final FeedbackService feedbackService;
    private final AttemptMapper attemptMapper;
    private final CodeExecutor primaryExecutor;
    private final CodeExecutor fallbackExecutor;

    public EvaluationServiceImpl(AttemptRepository attemptRepository,
                                  ProgressService progressService,
                                  FeedbackService feedbackService,
                                  AttemptMapper attemptMapper,
                                  @Qualifier("wandboxCodeExecutor") CodeExecutor primaryExecutor,
                                  @Qualifier("dockerCodeExecutorFallback") CodeExecutor fallbackExecutor) {
        this.attemptRepository = attemptRepository;
        this.progressService = progressService;
        this.feedbackService = feedbackService;
        this.attemptMapper = attemptMapper;
        this.primaryExecutor = primaryExecutor;
        this.fallbackExecutor = fallbackExecutor;
    }

    @Async("evaluationExecutor")
    @Retryable(retryFor = {RuntimeException.class}, backoff = @Backoff(delay = 1000))
    @Override
    public CompletableFuture<AttemptResponse> evaluateAttemptAsync(UUID attemptId) {
        log.info("=== ASYNC EVAL STARTED for attempt {} ===", attemptId);

        Attempt attempt = attemptRepository.findById(attemptId)
                .orElseThrow(() -> new CustomExceptions.ResourceNotFoundException("Attempt not found"));

        attempt.setStatus(Attempt.AttemptStatus.RUNNING);
        attempt = attemptRepository.save(attempt);
        log.info("Attempt {} set to RUNNING", attemptId);

        try {
            Problem problem = attempt.getProblem();
            List<Problem.TestCase> testCases = problem.getTestCases();
            Language language = attempt.getLanguage();

            log.info("Problem: {}, Language: {}, Test cases: {}",
                    problem.getId(), language, testCases == null ? 0 : testCases.size());

            if (testCases == null || testCases.isEmpty()) {
                log.warn("No test cases for problem {}", problem.getId());
                attempt.setResult(Attempt.AttemptResult.RUNTIME_ERROR);
                attempt.setStatus(Attempt.AttemptStatus.FAILED);
                attempt.setExecutionTime(0.0);
                attemptRepository.save(attempt);
                return CompletableFuture.completedFuture(attemptMapper.toResponse(attempt));
            }

            long totalExecutionTimeNanos = 0;
            Attempt.AttemptResult finalResult = Attempt.AttemptResult.ACCEPTED;

            for (int i = 0; i < testCases.size(); i++) {
                Problem.TestCase testCase = testCases.get(i);
                String input = testCase.getInput() == null ? "" : testCase.getInput();
                String expected = testCase.getExpectedOutput() == null ? "" : testCase.getExpectedOutput().trim();

                log.info("Running test case {}: input='{}', expected='{}'", i + 1, input, expected);

                ExecutionResult execResult = executeWithFallback(
                        attempt.getCode(),
                        input,
                        language,
                        5L
                );

                log.info("Execution result: success={}, timeout={}, output='{}', error='{}'",
                        execResult.success(), execResult.timeout(), execResult.output(), execResult.error());

                totalExecutionTimeNanos += execResult.executionTimeMs() * 1_000_000L;

                if (execResult.timeout()) {
                    finalResult = Attempt.AttemptResult.TIME_LIMIT_EXCEEDED;
                    log.warn("Test case {} timed out", i + 1);
                    break;
                }

                if (!execResult.success()) {
                    finalResult = Attempt.AttemptResult.COMPILATION_ERROR;
                    log.warn("Test case {} failed: {}", i + 1, execResult.error());
                    break;
                }

                String actualOutput = execResult.output() == null ? "" : execResult.output().trim();
                if (!actualOutput.equals(expected)) {
                    finalResult = Attempt.AttemptResult.WRONG_ANSWER;
                    log.warn("Test case {} wrong answer. Expected '{}', got '{}'", i + 1, expected, actualOutput);
                    break;
                }
            }

            double executionTimeSec = totalExecutionTimeNanos / 1_000_000_000.0;
            attempt.setExecutionTime(executionTimeSec);
            attempt.setResult(finalResult);
            attempt.setStatus(Attempt.AttemptStatus.COMPLETED);
            attempt = attemptRepository.save(attempt);
            log.info("Evaluation completed for attempt {} with result {}", attemptId, finalResult);

            progressService.updateAfterAttempt(attempt.getUser().getId(), attempt.getId());

            TransactionSynchronizationManager.registerSynchronization(new TransactionSynchronization() {
                @Override
                public void afterCommit() {
                    feedbackService.createForAttemptAsync(attemptId);
                }
            });

            return CompletableFuture.completedFuture(attemptMapper.toResponse(attempt));

        } catch (Exception ex) {
            log.error("!!! FATAL ERROR in async evaluation for attempt {} !!!", attemptId, ex);
            attempt.setStatus(Attempt.AttemptStatus.FAILED);
            attempt.setResult(Attempt.AttemptResult.RUNTIME_ERROR);
            attempt.setExecutionTime(0.0);
            attemptRepository.save(attempt);
            return CompletableFuture.completedFuture(attemptMapper.toResponse(attempt));
        }
    }

    private ExecutionResult executeWithFallback(String code, String input, Language language, long timeoutSeconds) {
        try {
            log.info("Attempting execution with Wandbox executor");
            ExecutionResult result = primaryExecutor.execute(code, input, language, timeoutSeconds);

            if (result.success() || (result.error() == null) || result.timeout()) {
                log.info("Wandbox execution successful");
                return result;
            }

            log.warn("Wandbox execution failed: {}. Falling back to Docker executor.", result.error());
            ExecutionResult fallbackResult = fallbackExecutor.execute(code, input, language, timeoutSeconds);
            log.info("Docker fallback execution result: success={}", fallbackResult.success());
            return fallbackResult;

        } catch (Exception e) {
            log.error("Exception in primary executor, falling back to Docker: {}", e.getMessage(), e);
            return fallbackExecutor.execute(code, input, language, timeoutSeconds);
        }
    }
}

                                      String firstName, String lastName) {

        Optional<Account> existing = accountRepository.findByUserEmailAndProvider(email, provider);
        if (existing.isPresent()) {
            Account account = existing.get();
            if (providerId != null && !providerId.isBlank() &&
                    (account.getProviderId() == null || account.getProviderId().isBlank())) {
                account.setProviderId(providerId);
                account.setEmailVerified(true);
                account = accountRepository.save(account);
            }
            return account;
        }

        // 2. Create user if needed
        User user = userRepository.findByEmail(email).orElseGet(() -> {
            User newUser = User.builder()
                    .firstName(firstName == null || firstName.isBlank() ? "User" : firstName)
                    .lastName(lastName)
                    .email(email)
                    .build();
            return userRepository.saveAndFlush(newUser);
        });

        // 3. Create and flush account directly
        Account newAccount = Account.builder()
                .provider(provider)
                .providerId(providerId)
                .role(Account.Role.USER)
                .emailVerified(true)
                .enabled(true)
                .accountNonLocked(true)
                .loginAttempts(0)
                .build();
        newAccount.setUser(user);   // set the user
        try {
            return accountRepository.saveAndFlush(newAccount);  // 🔥 immediate insert
        } catch (DataIntegrityViolationException e) {
            // Another thread inserted the account – fetch and return it
            log.warn("Duplicate account on insert for {} / {}", email, provider);
            return accountRepository.findByUserEmailAndProvider(email, provider)
                    .orElseThrow(() -> new CustomExceptions.BadRequestException("Failed to create account, please retry"));
        }
    }

    public void updateAfterAttempt(UUID userId, UUID attemptId) {
        User user = userRepository.findById(userId)
                .orElseThrow(() -> new CustomExceptions.ResourceNotFoundException("User not found"));
        Attempt attempt = attemptRepository.findById(attemptId)
                .orElseThrow(() -> new CustomExceptions.ResourceNotFoundException("Attempt not found"));

        if (!attempt.getUser().getId().equals(userId)) {
            throw new CustomExceptions.BadRequestException("Attempt does not belong to the current user");
        }

        boolean success = attempt.getResult() == Attempt.AttemptResult.ACCEPTED;
        Set<String> topics = resolveTopics(attempt.getProblem());
        UserProgressResponse first = null;

        for (String topic : topics) {
            UserProgress progress = userProgressRepository.findByUserIdAndTopic(userId, topic)
                    .orElseGet(() -> {
                        UserProgress created = new UserProgress();
                        created.setUser(user);
                        created.setTopic(topic);
                        return created;
                    });
            progress.addAttempt(success);
            UserProgress saved = userProgressRepository.save(progress);
            if (first == null) {
                first = userProgressMapper.toResponse(saved);
            }
        }

    }

    private Set<String> resolveTopics(Problem problem) {
        if (problem.getTags() == null || problem.getTags().isEmpty()) {
            return Set.of(problem.getTitle());
        }
        return new LinkedHashSet<>(problem.getTags());
    }

import com.shivansh.EliteCode.util.CodeSanitizer;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.springframework.web.client.RestClient;

import java.util.List;
import java.util.UUID;

@Service
@Slf4j
public class AiServiceImpl implements AiService {

    private final ProblemRepository problemRepository;
    private final RestClient restClient;

    @Value("${openrouter.api.key}")
    private String apiKey;

    @Value("${openrouter.model:openai/gpt-3.5-turbo}")
    private String model;

    @Value("${openrouter.api.url:https://openrouter.ai/api/v1/chat/completions}")
    private String apiUrl;

    // Customize temperature
    private static final double TEMPERATURE = 0.7;
    private static final int MAX_TOKENS = 1000;

    // Constructor-based RestClient building (or use @Bean)
    public AiServiceImpl(ProblemRepository problemRepository,
                         @Qualifier("openRouterRestClient") RestClient restClient) {
        this.problemRepository = problemRepository;
        this.restClient = restClient;
    }

    @Override
    public String ask(UUID problemId, Mode mode, String context) {
        // 1. Fetch problem statement if problemId provided
        String problemStatement = "";
        if (problemId != null) {
            Problem problem = problemRepository.findById(problemId)
                    .orElseThrow(() -> new CustomExceptions.ResourceNotFoundException("Problem not found"));
            problemStatement = problem.getOriginalText();
        }

        // 2. Build prompt using existing AiPromptBuilder
        String safeContext = CodeSanitizer.sanitizeCode(context);
        String userPrompt = AiPromptBuilder.buildPrompt(mode, problemStatement, safeContext);

        // 3. Prepare OpenRouter request
        ChatCompletionRequest request = new ChatCompletionRequest(
                model,
                List.of(new ChatCompletionRequest.Message("user", userPrompt)),
                MAX_TOKENS,
                TEMPERATURE
        );

        // 4. Call OpenRouter API
        try {
            ChatCompletionResponse response = restClient.post()
                    .uri(apiUrl)
                    .body(request)
                    .retrieve()
                    .onStatus(status -> status.is4xxClientError(), (req, res) -> {
                        String errorBody = new String(res.getBody().readAllBytes());
                        log.error("OpenRouter API error {}: {}", res.getStatusCode(), errorBody);
                        throw new CustomExceptions.BadRequestException("AI service error: " + errorBody);
                    })
                    .body(ChatCompletionResponse.class);

            if (response != null && response.choices() != null && !response.choices().isEmpty()) {
                String aiResponse = response.choices().get(0).message().content();
                log.info("AI response generated ({} tokens)", response.usage() != null ? response.usage().totalTokens() : 0);
                return aiResponse;
            } else {
                log.warn("Empty response from OpenRouter");
                return "I'm having trouble generating a response right now. Please try again.";
            }

        } catch (Exception e) {
            log.error("Failed to call OpenRouter", e);
            return "Sorry, the AI service is currently unavailable. Please try again later.";
        }
    }

    @Override
    public String generateSessionTitle(Mode mode, String problemTitle, String conversation) {
        String prompt = AiPromptBuilder.buildSessionTitlePrompt(
                mode,
                CodeSanitizer.sanitizeInput(problemTitle),
                CodeSanitizer.sanitizeInput(conversation)
        );

        ChatCompletionRequest request = new ChatCompletionRequest(
                model,
                List.of(new ChatCompletionRequest.Message("user", prompt)),
                40,
                0.2
        );

        try {
            ChatCompletionResponse response = restClient.post()
                    .uri(apiUrl)
                    .body(request)
                    .retrieve()
                    .onStatus(status -> status.is4xxClientError(), (req, res) -> {
                        String errorBody = new String(res.getBody().readAllBytes());
                        log.error("OpenRouter title error {}: {}", res.getStatusCode(), errorBody);
                        throw new CustomExceptions.BadRequestException("AI service error: " + errorBody);
                    })
                    .body(ChatCompletionResponse.class);

            if (response == null || response.choices() == null || response.choices().isEmpty()) {
                return fallbackTitle(mode, problemTitle);
            }

            return normalizeTitle(response.choices().get(0).message().content(), mode, problemTitle);
        } catch (Exception e) {
            log.error("Failed to generate session title", e);
            return fallbackTitle(mode, problemTitle);
        }
    }

    private String normalizeTitle(String title, Mode mode, String problemTitle) {
        if (title == null || title.isBlank()) {
            return fallbackTitle(mode, problemTitle);
        }

        String cleaned = title
                .replaceAll("[\\r\\n]+", " ")
                .replaceAll("^[\"'`*_\\s]+|[\"'`*_.!?:;\\s]+$", "")
                .replaceAll("\\s+", " ")
                .trim();

        if (cleaned.isBlank()) {
            return fallbackTitle(mode, problemTitle);
        }
        return cleaned.length() > 60 ? cleaned.substring(0, 60).trim() : cleaned;
    }

    private String fallbackTitle(Mode mode, String problemTitle) {
        if (problemTitle != null && !problemTitle.isBlank()) {
            return problemTitle.length() > 60 ? problemTitle.substring(0, 60).trim() : problemTitle.trim();
        }
        return switch (mode) {
            case INTERVIEW -> "Interview Practice";
            case LEARNING -> "Learning Practice";
            case STANDARD -> "Coding Help";
        };
    }
}

        } else if (hasTags) {
            problemPage = problemRepository.findByTagsIn(tags, pageable);
        } else {

        String payload = String.join("|",
                Objects.toString(title, ""),
                Objects.toString(originalText, ""),
                Objects.toString(difficulty, ""),
                tags == null ? "" : tags.stream().sorted().reduce("", (a, b) -> a + "," + b)
        );
        return HashUtil.sha256(payload);
    }

    @Override
    public TokenPair login(LoginRequest request) {
        User user = userRepository.findByEmail(request.email())
                .orElseThrow(() -> new CustomExceptions.ResourceNotFoundException("User not found"));

        Account account = accountRepository.findByUserEmailAndProvider(request.email(), Constants.LOCAL_PROVIDER)
                .orElseThrow(() -> new CustomExceptions.ResourceNotFoundException("Local account not found"));

        // Block login if email not verified
        if (!account.isEmailVerified()) {
            throw new CustomExceptions.BadRequestException("Please verify your email before logging in.");
        }

        if (account.isLocked()) {
            if (account.getLockedUntil() != null && account.getLockedUntil().isBefore(LocalDateTime.now())) {
                account.resetLoginAttempts();
                account.setAccountNonLocked(true);
            } else {
                log.warn("Login blocked: account locked for email {}", request.email());
                throw new CustomExceptions.BadRequestException("Account is locked");
            }
        }

        if (!passwordEncoder.matches(request.password(), account.getPasswordHash())) {
            account.incrementLoginAttempts();
            if (account.getLoginAttempts() >= 5) {
                account.lockAccount(Duration.ofMinutes(15));
            }
            accountRepository.save(account);
            log.warn("Login failed: invalid credentials for email {}", request.email());
            throw new CustomExceptions.BadRequestException("Invalid credentials");
        }

        verifyTotpIfEnabled(account, request.totpCode());

        account.resetLoginAttempts();
        account.setAccountNonLocked(true);
        account.setLastLoginAt(LocalDateTime.now());
        accountRepository.save(account);

        return issueTokenPair(user, account);
    }

    @Override
    public TokenPair refreshToken(String refreshToken) {
        // Validate token and extract claims
        if (!jwtTokenProvider.isTokenValid(refreshToken)) {
            throw new CustomExceptions.BadRequestException("Invalid refresh token");
        }
        Map<String, Object> claims = jwtTokenProvider.extractClaims(refreshToken);
        String tokenType = String.valueOf(claims.get("tokenType"));
        if (!Constants.REFRESH_TOKEN_TYPE.equalsIgnoreCase(tokenType)) {
            throw new CustomExceptions.BadRequestException("Invalid refresh token type");
        }

        String email = String.valueOf(claims.get("email"));
        String userId = String.valueOf(claims.get("userId"));
        User user = userRepository.findByEmail(email)
                .orElseThrow(() -> new CustomExceptions.ResourceNotFoundException("User not found"));
        if (!user.getId().toString().equals(userId)) {
            throw new CustomExceptions.BadRequestException("Refresh token does not match user");
        }

        Account account = accountRepository.findByUserEmailAndProvider(email, Constants.LOCAL_PROVIDER)
                .orElseThrow(() -> new CustomExceptions.ResourceNotFoundException("Local account not found"));

        if (!account.isEnabled() || account.isLocked()) {
            throw new CustomExceptions.BadRequestException("Account is disabled or locked");
        }
        if (!account.isEmailVerified()) {
            throw new CustomExceptions.BadRequestException("Email not verified");
        }

        String oldHash = hashToken(refreshToken);
        if (account.getRefreshTokenHash() == null || !account.getRefreshTokenHash().equals(oldHash)) {
            throw new CustomExceptions.BadRequestException("Refresh token has been rotated or revoked");
        }
        if (account.getRefreshTokenExpiresAt() == null || account.getRefreshTokenExpiresAt().isBefore(LocalDateTime.now())) {
            throw new CustomExceptions.BadRequestException("Refresh token expired");
        }

        // Issue new token pair (this automatically replaces the refresh token hash)
        return issueTokenPair(user, account);
    }

    private TokenPair issueTokenPair(User user, Account account) {
        Map<String, Object> claims = new HashMap<>();
        claims.put("userId", user.getId().toString());
        claims.put("email", user.getEmail());
        claims.put("provider", account.getProvider());
        claims.put("firstName", user.getFirstName());
        claims.put("lastName", user.getLastName());
        claims.put("role", account.getRole() == null ? Account.Role.USER.name() : account.getRole().name());
        claims.put("tokenType", Constants.ACCESS_TOKEN_TYPE);

        String accessToken = jwtTokenProvider.createToken(
                user.getEmail(),
                claims,
                Duration.ofMillis(jwtConfig.getExpirationMs())
        );

        long refreshTtl = jwtConfig.getRefreshExpirationMs() > 0
                ? jwtConfig.getRefreshExpirationMs()
                : Duration.ofDays(7).toMillis();
        Map<String, Object> refreshClaims = new HashMap<>();
        refreshClaims.put("userId", user.getId().toString());
        refreshClaims.put("email", user.getEmail());
        refreshClaims.put("tokenType", Constants.REFRESH_TOKEN_TYPE);
        String refreshToken = jwtTokenProvider.createToken(
                user.getEmail(),
                refreshClaims,
                Duration.ofMillis(refreshTtl)
        );

        // Store the new refresh token hash (overwrites the old one)
        account.setRefreshTokenHash(hashToken(refreshToken));
        account.setRefreshTokenExpiresAt(LocalDateTime.now().plusSeconds(refreshTtl / 1000));
        accountRepository.save(account);

        return new TokenPair(
                accessToken,
                "Bearer",
                jwtConfig.getExpirationMs() / 1000,
                refreshToken,
                user.getId(),
                user.getEmail(),
                user.getFirstName(),
                user.getLastName()
        );
    }

    private GoogleTokenInfo verifyGoogleToken(String idToken) {
        if (idToken == null || idToken.isBlank()) {
            throw new CustomExceptions.BadRequestException("idToken missing");
        }
        try {
            String url = googleTokenInfoUrl + "?id_token=" + idToken;
            JsonNode node = getJson(url, null);
            String email = node.path("email").asText(null);
            boolean emailVerified = node.path("email_verified").asBoolean(false);
            String aud = node.path("aud").asText(null);
            String sub = node.path("sub").asText(null);
            String givenName = node.path("given_name").asText(null);
            String familyName = node.path("family_name").asText(null);

            if (!emailVerified) {
                throw new CustomExceptions.BadRequestException("Google email not verified");
            }
            if (!isAllowedGoogleClient(aud)) {
                throw new CustomExceptions.BadRequestException("Invalid Google client");
            }
            return new GoogleTokenInfo(sub, email, givenName, familyName);
        } catch (CustomExceptions.BadRequestException ex) {
            throw ex;
        } catch (Exception ex) {
            throw new CustomExceptions.BadRequestException("Failed to verify Google token");
        }
    }

    private GithubProfile fetchGithubProfile(String accessToken) {
        if (accessToken == null || accessToken.isBlank()) {
            throw new CustomExceptions.BadRequestException("accessToken missing");
        }
        try {
            String userUrl = githubApiBase + "/user";
            JsonNode userNode = getJson(userUrl, Map.of(
                    "Authorization", "Bearer " + accessToken,
                    "Accept", "application/vnd.github+json"
            ));
            String id = userNode.path("id").asText(null);
            String name = userNode.path("name").asText(null);
            String login = userNode.path("login").asText(null);
            String email = userNode.path("email").asText(null);

            if (email == null || email.isBlank()) {
                String emailsUrl = githubApiBase + "/user/emails";
                JsonNode emailsNode = getJson(emailsUrl, Map.of(
                        "Authorization", "Bearer " + accessToken,
                        "Accept", "application/vnd.github+json"
                ));
                for (JsonNode entry : emailsNode) {
                    if (entry.path("primary").asBoolean(false) && entry.path("verified").asBoolean(false)) {
                        email = entry.path("email").asText(null);
                        break;
                    }
                }
            }

            if (email == null || email.isBlank()) {
                throw new CustomExceptions.BadRequestException("GitHub email not available");
            }

            String firstName = null;
            String lastName = null;
            if (name != null && !name.isBlank()) {
                String[] parts = name.trim().split("\\s+", 2);
                firstName = parts[0];
                if (parts.length > 1) {
                    lastName = parts[1];
                }
            } else if (login != null && !login.isBlank()) {
                firstName = login;
            }
            return new GithubProfile(id, email, firstName, lastName);
        } catch (CustomExceptions.BadRequestException ex) {
            throw ex;
        } catch (Exception ex) {
            throw new CustomExceptions.BadRequestException("Failed to verify GitHub token");
        }
    }

    private JsonNode getJson(String url, Map<String, String> headers) throws Exception {
        HttpRequest.Builder builder = HttpRequest.newBuilder()
                .uri(URI.create(url))
                .GET();
        if (headers != null) {
            headers.forEach(builder::header);
        }
        HttpResponse<String> response = httpClient.send(builder.build(), HttpResponse.BodyHandlers.ofString());
        if (response.statusCode() < 200 || response.statusCode() >= 300) {
            throw new IllegalStateException("Request failed: " + response.statusCode());
        }
        return objectMapper.readTree(response.body());
    }

    private boolean isAllowedGoogleClient(String aud) {
        if (aud == null || aud.isBlank()) {
            return false;
        }
        if (googleClientIds == null || googleClientIds.isBlank()) {
            return false;
        }
        String[] allowed = googleClientIds.split(",");
        for (String clientId : allowed) {
            if (aud.equals(clientId.trim())) {
                return true;
            }
        }
        return false;
    }

    @Override
    @Transactional
    public void changePassword(UUID userId, String currentPassword, String newPassword) {
        Account account = accountRepository.findByUserIdAndProvider(userId, Constants.LOCAL_PROVIDER)
                .orElseThrow(() -> new CustomExceptions.ResourceNotFoundException("Local account not found"));
        if (!passwordEncoder.matches(currentPassword, account.getPasswordHash())) {
            throw new CustomExceptions.BadRequestException("Current password is incorrect");
        }
        account.setPasswordHash(passwordEncoder.encode(newPassword));
        account.setPasswordChangedAt(LocalDateTime.now());
        // Invalidate all refresh tokens for this account
        account.setRefreshTokenHash(null);
        account.setRefreshTokenExpiresAt(null);
        accountRepository.save(account);
    }

    private record GoogleTokenInfo(String providerId, String email, String firstName, String lastName) {
    }

    private record GithubProfile(String providerId, String email, String firstName, String lastName) {
    }
}

package com.shivansh.EliteCode.exception;

import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.time.Instant;
import java.util.HashMap;
import java.util.Map;

@RestControllerAdvice
@Slf4j
public class GlobalExceptionHandler {

    @ExceptionHandler(CustomExceptions.ResourceNotFoundException.class)
    public ResponseEntity<ErrorResponse> handleNotFound(CustomExceptions.ResourceNotFoundException ex) {
        return buildErrorResponse(ex.getMessage(), HttpStatus.NOT_FOUND);
    }

    @ExceptionHandler(CustomExceptions.BadRequestException.class)
    public ResponseEntity<ErrorResponse> handleBadRequest(CustomExceptions.BadRequestException ex) {
        return buildErrorResponse(ex.getMessage(), HttpStatus.BAD_REQUEST);
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ErrorResponse> handleValidation(MethodArgumentNotValidException ex) {
        Map<String, String> errors = new HashMap<>();
        ex.getBindingResult().getFieldErrors().forEach(error ->
                errors.put(error.getField(), error.getDefaultMessage())
        );
        return ResponseEntity.badRequest().body(
                new ErrorResponse(HttpStatus.BAD_REQUEST.value(), "Validation failed", Instant.now().toString(), errors)
        );
    }

    @ExceptionHandler(AccessDeniedException.class)
    public ResponseEntity<ErrorResponse> handleAccessDenied(AccessDeniedException ex) {
        return buildErrorResponse("Access denied", HttpStatus.FORBIDDEN);
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ErrorResponse> handleGeneric(Exception ex) {
        log.error("Unexpected error", ex);
        return buildErrorResponse("An internal error occurred", HttpStatus.INTERNAL_SERVER_ERROR);
    }

    private ResponseEntity<ErrorResponse> buildErrorResponse(String message, HttpStatus status) {
        return ResponseEntity.status(status).body(
                new ErrorResponse(status.value(), message, Instant.now().toString(), null)
        );
    }

    public record ErrorResponse(int status, String message, String timestamp, Map<String, String> details) {}


}
