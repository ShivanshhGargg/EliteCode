export const formatDate = (dateString) => {
  if (!dateString) return 'N/A'
  return new Date(dateString).toLocaleDateString('en-US', {
    year: 'numeric', month: 'short', day: 'numeric',
    hour: '2-digit', minute: '2-digit',
  })
}

export const formatTime = (ms) => {
  if (!ms) return 'N/A'
  return `${ms}ms`
}

export const formatSuccessRate = (rate) => {
  if (rate == null) return '0%'
  return `${(rate * 100).toFixed(1)}%`
}

export const truncate = (text, length = 100) => {
  if (!text) return ''
  return text.length > length ? text.substring(0, length) + '...' : text
}

export const getDifficultyColor = (difficulty) => {
  switch (difficulty) {
    case 'EASY': return 'text-neon-green'
    case 'MEDIUM': return 'text-neon-orange'
    case 'HARD': return 'text-neon-red'
    default: return 'text-txt-muted'
  }
}

export const getDifficultyBg = (difficulty) => {
  switch (difficulty) {
    case 'EASY': return 'bg-neon-green/10 border-neon-green/20 text-neon-green'
    case 'MEDIUM': return 'bg-neon-orange/10 border-neon-orange/20 text-neon-orange'
    case 'HARD': return 'bg-neon-red/10 border-neon-red/20 text-neon-red'
    default: return 'bg-white/5 border-white/10 text-txt-muted'
  }
}

export const getResultColor = (result) => {
  switch (result) {
    case 'ACCEPTED': return 'text-neon-green'
    case 'WRONG_ANSWER': return 'text-neon-red'
    case 'RUNTIME_ERROR': return 'text-neon-orange'
    case 'TIME_LIMIT_EXCEEDED': return 'text-neon-yellow'
    case 'PENDING':
    case 'RUNNING': return 'text-cyan'
    default: return 'text-txt-muted'
  }
}

export const getModeDescription = (mode) => {
  const descriptions = {
    STANDARD: 'Balanced assistant guidance with clear explanations and execution support',
    INTERVIEW: 'Sharper, question-first coaching that behaves like a technical interviewer',
    LEARNING: 'Progressive tutoring with step-by-step hints and concept breakdowns',
  }
  return descriptions[mode] || descriptions.STANDARD
}

export const getErrorMessage = (error) => {
  if (!error) return 'An unknown error occurred'
  if (error.response?.data?.message) return error.response.data.message
  if (error.response?.data?.details) return error.response.data.details
  if (error.message) return error.message
  if (typeof error === 'string') return error
  return 'An unknown error occurred'
}

import { useState, useEffect, useRef } from 'react'
import { Play, Bug, CheckCircle, ChevronDown, Clock, Cpu, Minimize2, Maximize2, GripHorizontal, Loader2, Terminal } from 'lucide-react'
import { useMutation } from '@tanstack/react-query'
import { attemptService } from '../api/attemptService'
import { hintService } from '../api/hintService'
import useChatStore from '../store/chatStore'
import useToastStore from '../store/toastStore'
import useWorkspaceStore from '../store/workspaceStore'
import { CodeEditor } from './CodeEditor'
import { NeonButton } from './NeonButton'
import { usePollAttempt } from '../hooks/usePollAttempt'
import { getErrorMessage } from '../utils/error'
import { formatTime } from '../utils/format'

const STARTER_CODE = {
  JAVA: `class Solution {
    public int solve(int[] nums) {
        // Write your solution here
        return 0;
    }
}`,
  PYTHON: `class Solution:
    def solve(self, nums):
        # Write your solution here
        return 0`,
  JAVASCRIPT: `function solve(nums) {
    // Write your solution here
    return 0;
}`,
  CPP: `class Solution {
public:
    int solve(vector<int>& nums) {
        // Write your solution here
        return 0;
    }
};`,
}

const LANGUAGES = [
  { value: 'JAVA', label: 'Java' },
  { value: 'PYTHON', label: 'Python' },
  { value: 'JAVASCRIPT', label: 'JavaScript' },
  { value: 'CPP', label: 'C++' },
]

export function CodeWorkspace({ workspaceControls = null }) {
  const [code, setCode] = useState(STARTER_CODE.JAVA)
  const [language, setLanguage] = useState('JAVA')
  const [attemptId, setAttemptId] = useState(null)
  const [feedback, setFeedback] = useState(null)
  const [resultsHeight, setResultsHeight] = useState(35) // percentage of available space
  const [isResultsMinimized, setIsResultsMinimized] = useState(false)
  const isDraggingHeight = useRef(false)
  const containerRef = useRef(null)

  const { mode, currentSessionId, markSessionWork, updateSessionTitle } = useChatStore()
  const { addToast } = useToastStore()
  const { selectedProblem } = useWorkspaceStore()

  const { mutate: submit, isPending: isSubmitting } = useMutation({
    mutationFn: () => {
      if (!selectedProblem?.id) {
        throw new Error('Select a problem before running code')
      }
      return attemptService.submitAttempt(
          selectedProblem.id,
          mode,
          code,
          '',
          language
      )
    },
    onSuccess: (data) => {
      setAttemptId(data.id)
      setFeedback(null)
      if (currentSessionId) {
        markSessionWork(currentSessionId)
        hintService.generateSessionTitle({
          mode,
          problemTitle: selectedProblem?.title || '',
          conversation: `${language} code run submitted for ${selectedProblem?.title || 'a coding problem'}\n${code.slice(0, 1200)}`,
        }).then((result) => {
          if (result?.title) updateSessionTitle(currentSessionId, result.title)
        }).catch(() => {})
      }
    },
    onError: (err) => addToast(getErrorMessage(err), 'error'),
  })

  const { attempt, isLoading: polling } = usePollAttempt(attemptId, async (data) => {
    if (data.status === 'FAILED') {
      addToast('Some tests failed', 'error')
    }
    try {
      const feedbackData = await attemptService.getAttemptFeedback(data.id)
      setFeedback(feedbackData)
    } catch (err) {
      console.error('Feedback fetch failed', err)
    }
  })

  useEffect(() => {
    setCode(STARTER_CODE[language])
  }, [language])

  const handleLanguageChange = (newLang) => {
    setLanguage(newLang)
    setCode(STARTER_CODE[newLang])
  }

  const handleDebug = () => {
    if (!selectedProblem?.id) {
      return
    }
    submit()
  }

  const handleSubmitAttempt = () => {
    if (!selectedProblem?.id) {
      return
    }
    submit()
  }

  const isLearnMode = mode === 'LEARNING'
  const timeComplexity = feedback?.timeComplexity || attempt?.timeComplexity || '—'
  const spaceComplexity = feedback?.spaceComplexity || attempt?.spaceComplexity || '—'
  const testResults = feedback?.testResults || attempt?.testResults || []
  const isLoading = isSubmitting || polling

  const handleMouseMove = (e) => {
    if (!isDraggingHeight.current) return
    const container = containerRef.current
    if (!container) return
    const rect = container.getBoundingClientRect()
    const toolbar = document.getElementById('code-workspace-toolbar')
    const toolbarHeight = toolbar?.offsetHeight || 48
    const availableHeight = rect.height - toolbarHeight
    const mouseYRelative = e.clientY - rect.top - toolbarHeight
    let newHeight = ((availableHeight - mouseYRelative) / availableHeight) * 100
    newHeight = Math.min(Math.max(newHeight, 15), 70)
    setResultsHeight(newHeight)
  }

  const handleMouseUp = () => {
    isDraggingHeight.current = false
    document.removeEventListener('mousemove', handleMouseMove)
    document.removeEventListener('mouseup', handleMouseUp)
  }

  const startResize = (e) => {
    e.preventDefault()
    isDraggingHeight.current = true
    document.addEventListener('mousemove', handleMouseMove)
    document.addEventListener('mouseup', handleMouseUp)
  }

  // Calculate editor height based on results panel state
  const getEditorHeight = () => {
    if (isResultsMinimized) return '100%'
    return `calc(100% - ${resultsHeight}% - 12px)` // 12px for gap + handle
  }

  return (
      <div ref={containerRef} className="leetcode-panel flex h-full min-h-0 flex-col overflow-hidden">
        <div id="code-workspace-toolbar" className="shrink-0 flex items-center justify-between px-3 py-2 border-b border-white/10 bg-white/[0.02]">
          <div className="flex items-center gap-3">
            <div className="flex items-center gap-2 text-sm font-semibold text-txt-primary">
              <Terminal size={15} className="text-cyan" />
              Code
            </div>
            <div className="relative">
              <select
                  value={language}
                  onChange={(e) => handleLanguageChange(e.target.value)}
                  className="appearance-none bg-black/40 border border-white/15 rounded-md px-3 py-1.5 pr-8 text-sm text-txt-primary cursor-pointer hover:bg-black/60 focus:outline-none focus:ring-1 focus:ring-cyan/50"
              >
                {LANGUAGES.map((lang) => (
                    <option key={lang.value} value={lang.value}>
                      {lang.label}
                    </option>
                ))}
              </select>
              <ChevronDown size={12} className="absolute right-2.5 top-1/2 -translate-y-1/2 text-txt-muted pointer-events-none" />
            </div>
          </div>

          <div className="flex items-center gap-2">
            <NeonButton variant="ghost" size="sm" onClick={handleDebug} disabled={isLoading}>
              <Bug size={12} className="mr-1" /> Debug
            </NeonButton>
            {isLearnMode && (
                <NeonButton variant="solid" size="sm" loading={isLoading} disabled={!selectedProblem?.id || isLoading} onClick={handleSubmitAttempt}>
                  <Play size={12} className="mr-1" /> Run
                </NeonButton>

import { useState } from 'react'
import { X } from 'lucide-react'
import { NeonButton } from './NeonButton'
import useInterviewStore from '../store/interviewStore'

export function LearningTimerModal({ isOpen, onClose, sessionId }) {
    const [hours, setHours] = useState(0)
    const [minutes, setMinutes] = useState(0)
    const { getSession, startSession, updateTimer } = useInterviewStore()
    const totalSeconds = hours * 3600 + minutes * 60

    if (!isOpen) return null

    const handleSetTimer = () => {
        if (totalSeconds <= 0) {
            return
        }
        const session = getSession(sessionId)
        if (session?.started) {
            // If session already started, update its time limit and remaining
            updateTimer(sessionId, session.elapsed, totalSeconds)
        } else {
            // Start session with timer
            startSession(sessionId, { mode: 'LEARNING', timeLimit: totalSeconds })
        }
        onClose()
    }

    return (
        <div className="fixed inset-0 z-[200] flex items-center justify-center p-4">
            <div className="absolute inset-0 bg-void/80 backdrop-blur-sm" onClick={onClose} />
            <div className="relative w-full max-w-md glass-panel rounded-2xl overflow-hidden shadow-2xl">
                <div className="p-5 border-b border-white/10 flex items-center justify-between">
                    <h2 className="text-lg font-semibold text-txt-primary">Set Learning Timer</h2>
                    <button onClick={onClose} className="p-1 rounded-lg hover:bg-white/10 text-txt-muted">
                        <X size={18} />
                    </button>

                <div className="p-5 space-y-5">
                    <div className="flex gap-4 justify-center">

                            <label className="block text-xs text-txt-muted mb-2">Hours</label>

export function TwoFactorSetup() {
    const [step, setStep] = useState('init'); // init, showQR, verify
    const [secret, setSecret] = useState('');
    const [otpauthUrl, setOtpauthUrl] = useState('');
    const [code, setCode] = useState('');
    const { addToast } = useToastStore();

    const setupMutation = useMutation({
        mutationFn: () => authService.setupTwoFactor(),
        onSuccess: (data) => {
            setSecret(data.secret);
            setOtpauthUrl(data.otpauthUrl);
            setStep('showQR');
        },
        onError: (err) => addToast(err.message, 'error'),
    });

    const enableMutation = useMutation({
        mutationFn: (code) => authService.enableTwoFactor(code),
        onSuccess: () => {
            addToast('2FA enabled successfully', 'success');
            setStep('enabled');
        },
        onError: (err) => addToast(err.message, 'error'),
    });

    const disableMutation = useMutation({
        mutationFn: (code) => authService.disableTwoFactor(code),
        onSuccess: () => {
            addToast('2FA disabled', 'success');
            setStep('init');
        },
        onError: (err) => addToast(err.message, 'error'),
    });

    if (step === 'init') {
        return (
            <div>
                <NeonButton onClick={() => setupMutation.mutate()}>Set up 2FA</NeonButton>
                {disableMutation.isIdle && (
                    <NeonButton variant="ghost" onClick={() => setStep('disable')} className="ml-2">Disable 2FA</NeonButton>
                )}
            </div>
        );
    }

    if (step === 'showQR') {
        return (
            <div className="space-y-4">
                <p>Scan this QR code with Google Authenticator or any TOTP app:</p>
                <QRCodeSVG value={otpauthUrl} size={200} />
                <p>Or manually enter the secret: <code className="bg-black/30 p-1 rounded">{secret}</code></p>
                <input
                    type="text"
                    placeholder="Enter 6-digit code"
                    value={code}
                    onChange={(e) => setCode(e.target.value)}
                    className="input-neon w-48"
                />
                <NeonButton onClick={() => enableMutation.mutate(code)} loading={enableMutation.isPending}>
                    Verify and Enable
                </NeonButton>
            </div>
        );
    }

    if (step === 'disable') {
        return (
            <div className="space-y-4">
                <p>Enter your current TOTP code to disable 2FA:</p>
                <input
                    type="text"
                    placeholder="6-digit code"
                    value={code}
                    onChange={(e) => setCode(e.target.value)}
                    className="input-neon w-48"
                />
                <NeonButton onClick={() => disableMutation.mutate(code)} variant="magenta" loading={disableMutation.isPending}>
                    Disable 2FA
                </NeonButton>
            </div>
        );
    }

    return <div>2FA is enabled. Your account is protected.</div>;
}

  string: 'Pattern matching, parsing, frequency maps, and two-pointer scans.',

  'two pointers': 'Opposing or same-direction scans that shrink brute force.',

  backtracking: 'Explore choices, prune invalid branches, and restore state.',

const DEFAULT_TOPICS = [
  'Array',
  'String',
  'Hash Table',
  'Two Pointers',
  'Sliding Window',
  'Stack',
  'Binary Search',
  'Tree',
  'Graph',
  'Dynamic Programming',
  'Greedy',
  'Backtracking',
]

const normalize = (value = '') => value.toLowerCase().replace(/[-_]/g, ' ').trim()

const topicIntro = (topic) => {
  const key = normalize(topic)
  const matched = Object.keys(TOPIC_INTROS).find((item) => key.includes(item))
  return matched ? TOPIC_INTROS[matched] : 'A focused track of problems that build one reusable interview pattern.'
}

const pickSuggestions = (problems, selectedTopic) => {
  const scoped = selectedTopic === 'All'
      ? problems
      : problems.filter((problem) => problem.tags?.some((tag) => normalize(tag) === normalize(selectedTopic)))
  return [...scoped]
      .sort((a, b) => {
        const rank = { EASY: 0, MEDIUM: 1, HARD: 2 }
        return (rank[a.difficulty] ?? 1) - (rank[b.difficulty] ?? 1)
      })
      .slice(0, 4)
}

export function LearningWelcome() {
  const [selectedTopic, setSelectedTopic] = useState('All')
  const { addToast } = useToastStore()
  const { setSelectedProblem } = useWorkspaceStore()
  const { sessions, currentSessionId, createSession, setCurrentSession, addMessage, updateSessionTitle } = useChatStore()

  const { data, isLoading, error } = useQuery({
    queryKey: ['learning-problems'],
    queryFn: () => problemService.getProblems(0, 100),
  })

  const problems = data?.content || []
  const topics = useMemo(() => {
    const counts = new Map()
    problems.forEach((problem) => {
      ;(problem.tags || []).forEach((tag) => {
        const label = tag.trim()
        if (label) counts.set(label, (counts.get(label) || 0) + 1)
      })
    })
    const loadedTopics = [...counts.entries()].sort((a, b) => b[1] - a[1]).map(([topic]) => topic)
    return ['All', ...(loadedTopics.length ? loadedTopics : DEFAULT_TOPICS)]
  }, [problems])

  const visibleProblems = useMemo(() => {
    if (selectedTopic === 'All') return problems
    return problems.filter((problem) => problem.tags?.some((tag) => normalize(tag) === normalize(selectedTopic)))
  }, [problems, selectedTopic])

  const ensureLearningSession = () => {
    const current = sessions.find((session) => session.id === currentSessionId && session.mode === 'LEARNING')
    if (current) return current.id

    const existing = sessions.find((session) => session.mode === 'LEARNING')
    if (existing) {
      setCurrentSession(existing.id)
      return existing.id
    }

    return createSession(null, 'Learning Assistant', 'LEARNING')
  }

    mutationFn: async () => {
      const suggestions = pickSuggestions(problems, selectedTopic)

const readFileHeader = (file) =>
    new Promise((resolve) => {
      const reader = new FileReader()
      reader.onload = () => {
        const result = reader.result
        if (!(result instanceof ArrayBuffer)) {
          resolve('')
          return
        }
        const arr = new Uint8Array(result)
        const header = Array.from(arr.slice(0, 8))
            .map((b) => b.toString(16).padStart(2, '0'))
            .join('')
        resolve(header)
      }
      reader.readAsArrayBuffer(file.slice(0, 8))
    })

const isArchiveByHeader = (header) => {
  return header.startsWith('504b0304') || header.startsWith('52617221') || header.startsWith('377abcaf')

const isFileAllowed = async (file) => {
  const ext = file.name.substring(file.name.lastIndexOf('.')).toLowerCase()
  if (FORBIDDEN_EXTENSIONS.includes(ext)) return false
  if (file.type && file.type.startsWith('image/')) return false
  const header = await readFileHeader(file)
  return !isArchiveByHeader(header)
}

const readFileAsText = (file) =>

      const reader = new FileReader()
      reader.onload = () => resolve(reader.result || '')
      reader.onerror = reject
      reader.readAsText(file)
    })

import { useEffect, useState } from 'react'
import { Clock } from 'lucide-react'
import useInterviewStore from '../store/interviewStore'

export function ModeTimer({ mode, sessionId }) {
    const { getSession, updateTimer, deductPoints, endSession } = useInterviewStore()
    const sessionData = getSession(sessionId)
    const [remaining, setRemaining] = useState(null)
    const [elapsed, setElapsed] = useState(0)

    useEffect(() => {
        if (!sessionData?.started || sessionData?.paused) return

        const interval = setInterval(() => {
            const now = Date.now()
            const elapsedSec = Math.floor((now - sessionData.startTime) / 1000)
            setElapsed(elapsedSec)

            if (mode === 'INTERVIEW' && sessionData.timeLimit) {
                const rem = Math.max(0, sessionData.timeLimit - elapsedSec)
                setRemaining(rem)
                updateTimer(sessionId, elapsedSec, rem)
                if (rem === 0) {
                    window.dispatchEvent(new CustomEvent('elitecode-time-up', {
                        detail: { sessionId, finalScore: sessionData.score }
                    }))
                    endSession(sessionId)
                    clearInterval(interval)
                }
            } else if (mode === 'LEARNING' && sessionData.timeLimit) {
                const rem = Math.max(0, sessionData.timeLimit - elapsedSec)
                setRemaining(rem)
                updateTimer(sessionId, elapsedSec, rem)
                if (rem === 0) {
                    window.dispatchEvent(new CustomEvent('elitecode-time-up', {
                        detail: { sessionId, finalScore: null }
                    }))
                    clearInterval(interval)
                }
            } else {
                // Unlimited session – just update elapsed
                updateTimer(sessionId, elapsedSec, null)
            }
        }, 1000)

        return () => clearInterval(interval)
    }, [sessionData?.started, sessionData?.paused, sessionId, mode, sessionData?.timeLimit, sessionData?.score, updateTimer, endSession])

    // Subscribe to point deductions for interview
    useEffect(() => {
        const handler = (e) => {
            if (mode === 'INTERVIEW' && sessionData?.started) {
                deductPoints(sessionId, e.detail || 10)
            }
        }
        window.addEventListener('elitecode-deduct-points', handler)
        return () => window.removeEventListener('elitecode-deduct-points', handler)
    }, [mode, sessionId, sessionData?.started, deductPoints])

    const formatTime = (sec) => {
        if (sec === null || sec === undefined) return '--:--'
        const hours = Math.floor(sec / 3600)
        const minutes = Math.floor((sec % 3600) / 60)
        const seconds = sec % 60
        if (hours > 0) {
            return `${hours}h ${minutes.toString().padStart(2, '0')}m`
        }
        return `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`
    }

    if (!sessionData?.started) return null

    let displayText = ''
    if (mode === 'INTERVIEW') {
        displayText = formatTime(remaining !== null ? remaining : sessionData.remaining)
        return (
            <div className="flex items-center gap-2 text-sm font-mono">
                <div className="flex items-center gap-1.5 text-neon-red">
                    <Clock size={14} />
                    <span>{displayText}</span>
                </div>
                <div className="flex items-center gap-1.5 text-cyan">
                    <span>Score: {sessionData.score}</span>
                </div>
            </div>
        )
    } else if (mode === 'LEARNING') {
        if (sessionData.timeLimit) {
            displayText = formatTime(remaining !== null ? remaining : sessionData.remaining)
        } else {
            displayText = formatTime(elapsed)
        }
        return (
            <div className="flex items-center gap-1.5 text-sm font-mono text-cyan">
                <Clock size={14} />
                <span>{displayText}</span>
            </div>
        )
    }

    return null
}

import { useEffect, useState } from 'react'
import axios from 'axios'
import useAuthStore from '../store/authStore'
import {authService} from "../api/authService.js";

const API_BASE_URL = import.meta.env.VITE_API_BASE_URL || 'http://localhost:8080/api'
let hydrationPromise = null

const readUser = () => {
  try {
    const storedUser = localStorage.getItem('user')
    return storedUser ? JSON.parse(storedUser) : null
  } catch {
    return null
  }
}

const getAccessToken = (payload) =>
  payload?.accessToken || payload?.access_token || payload?.token || payload?.data?.accessToken || null

const getUserFromAuthPayload = (payload, fallbackUser) => {
  const source = payload?.data || payload?.user || payload || {}
  return {
    ...fallbackUser,
    id: source.userId || source.id || fallbackUser?.id,
    email: source.email || fallbackUser?.email,
    firstName: source.firstName ?? fallbackUser?.firstName,
    lastName: source.lastName ?? fallbackUser?.lastName,
  }
}

export const hydrateAuth = async () => {
  if (hydrationPromise) return hydrationPromise
  hydrationPromise = (async () => {
    const { setAuth, clearAuth, logout, setLoading } = useAuthStore.getState()
    setLoading(true)
    const storedUser = readUser()
    if (!storedUser) {
      clearAuth()
      setLoading(false)
      hydrationPromise = null
      return
    }

    setAuth(storedUser, null)

    try {
      const { data } = await axios.post(`${API_BASE_URL}/auth/refresh`, {}, { withCredentials: true })
      const newAccessToken = getAccessToken(data)
      if (newAccessToken) {
        const refreshedUser = getUserFromAuthPayload(data, storedUser)
        setAuth(refreshedUser, newAccessToken)
        localStorage.setItem('user', JSON.stringify(refreshedUser))
      } else {
        logout()
        localStorage.removeItem('user')
      }
    } catch {
      logout()
      localStorage.removeItem('user')
    } finally {
      setLoading(false)
      hydrationPromise = null
    }
  })()
  return hydrationPromise
}

export function useAuth() {
  const { user, accessToken, isLoading, error, setAuth, clearAuth, logout } = useAuthStore()
  const [isHydrated, setIsHydrated] = useState(false)

  useEffect(() => {
    let active = true
    hydrateAuth().finally(() => {
      if (active) setIsHydrated(true)
    })
    return () => { active = false }
  }, [])

  const login = (user, accessToken) => {
    setAuth(user, accessToken)
    localStorage.setItem('user', JSON.stringify(user))
  }

  const handleLogout = async () => {
    try {
      await authService.logout();
    } catch (e) { /* ignore */ }
    logout();
    localStorage.removeItem('user');
  };

  return {
    user, accessToken, isLoading: !isHydrated || isLoading, error,
    isAuthenticated: !!accessToken && isHydrated, isHydrated,
    login, logout: handleLogout, setAuth, clearAuth,
  }
}

import { useEffect, useRef } from 'react'
import { useQuery } from '@tanstack/react-query'
import { attemptService } from '../api/attemptService'

const POLL_INTERVAL = parseInt(import.meta.env.VITE_POLL_INTERVAL || '2000')

export function usePollAttempt(attemptId, onComplete) {
  const completedRef = useRef(false)

  const { data: attempt, isLoading, error } = useQuery({
    queryKey: ['attempt', attemptId],
    queryFn: () => attemptService.getAttemptById(attemptId),
    enabled: !!attemptId,
    refetchInterval: (query) => {
      const data = query?.state?.data
      if (data && (data.status === 'COMPLETED' || data.status === 'FAILED')) {
        if (!completedRef.current && onComplete) {
          completedRef.current = true
          onComplete(data)
        }
        return false
      }
      return POLL_INTERVAL
    },
  })

  useEffect(() => {
    return () => {
      completedRef.current = false
    }
  }, [attemptId])

  return { attempt, isLoading, error, isPolling: !!attemptId && (!attempt || (attempt.status !== 'COMPLETED' && attempt.status !== 'FAILED')) }
}

import axios from 'axios'
import useAuthStore from '../store/authStore'

const API_BASE_URL = import.meta.env.VITE_API_BASE_URL || 'http://localhost:8080/api'

const axiosClient = axios.create({
  baseURL: API_BASE_URL,
  headers: { 'Content-Type': 'application/json' },
  withCredentials: true,
})

let isRefreshing = false
let refreshSubscribers = []

const extractAccessToken = (payload) =>
    payload?.accessToken || payload?.access_token || payload?.token || payload?.data?.accessToken || null

function onRefreshed(token) {
  refreshSubscribers.forEach((cb) => cb(token))
  refreshSubscribers = []
}

function addSubscriber(cb) {
  refreshSubscribers.push(cb)
}

axiosClient.interceptors.request.use(
    (config) => {
      const token = useAuthStore.getState().accessToken
      if (token) {
        config.headers.Authorization = `Bearer ${token}`
      }
      return config
    },
    (error) => Promise.reject(error)
)

axiosClient.interceptors.response.use(
    (res) => res,
    async (error) => {
      const original = error.config
      // ✅ Also treat 403 as auth error
      if ((error.response?.status === 401 || error.response?.status === 403) && !original._retry) {
        original._retry = true
        if (isRefreshing) {
          return new Promise((resolve) => {
            addSubscriber((token) => {
              original.headers.Authorization = `Bearer ${token}`
              resolve(axiosClient(original))
            })
          })
        }
        isRefreshing = true
        try {
          const { data } = await axios.post(`${API_BASE_URL}/auth/refresh`, {}, { withCredentials: true })
          const newAccessToken = extractAccessToken(data)
          if (!newAccessToken) {
            throw new Error('Missing access token in refresh response')
          }
          // Update store and localStorage
          const currentUser = useAuthStore.getState().user
          useAuthStore.getState().setAuth(currentUser, newAccessToken)
          onRefreshed(newAccessToken)
          original.headers.Authorization = `Bearer ${newAccessToken}`
          return axiosClient(original)
        } catch (refreshError) {
          // Refresh failed – force logout
          useAuthStore.getState().logout()
          localStorage.removeItem('user')
          // Avoid redirect loop: only redirect if not already on login page
          if (!window.location.pathname.includes('/login')) {
            window.location.href = '/login'
          }
          return Promise.reject(refreshError)
        } finally {
          isRefreshing = false
        }
      }
      return Promise.reject(error)
    }
)

export default axiosClient

                                <Calendar size={16} className="text-txt-muted" />
                                <div>
                                    <p className="text-xs text-txt-muted uppercase tracking-[0.24em]">Joined</p>
                                    <p className="text-sm text-txt-primary">{user?.createdAt ? formatDate(user.createdAt) : 'N/A'}</p>
                                </div>
                            </div>
                        </div>
                    )}
                </div>

                <div className="glass-panel rounded-[32px] p-6 md:p-8 mt-6">
                    <div className="flex items-center justify-between mb-6">
                        <h2 className="text-lg font-semibold text-txt-primary">Appearance</h2>
                        <span className="text-[10px] uppercase tracking-[0.28em] text-txt-muted">Themes</span>
                    </div>
                    <ThemeSwitcher />
                </div>

                <div className="glass-panel rounded-[32px] p-6 md:p-8 mt-6">
                    <h2 className="text-lg font-semibold text-txt-primary mb-6">Security</h2>
                    <TwoFactorSetup />
                </div>
            </div>

            <div className="glass-panel rounded-[32px] p-6 mt-6">
                <h2 className="text-lg font-semibold mb-4">Linked Accounts</h2>
                {accounts?.length === 0 && <p className="text-txt-muted">No linked accounts</p>}
                {accounts?.map(acc => (
                    <div key={acc.id} className="flex justify-between items-center py-2 border-b border-white/5">
                        <span>{acc.provider} ({acc.email})</span>
                        {acc.provider !== 'local' && (
                            <NeonButton size="sm" variant="ghost" onClick={() => unlinkMutation.mutate(acc.id)}>Unlink</NeonButton>
                        )}
                    </div>
                ))}
            </div>
        </div>
    )
}

import { create } from 'zustand';
import { persist } from 'zustand/middleware';

const useInterviewStore = create(persist((set, get) => ({
    sessions: {},

    startSession: (sessionId, options = {}) => {
        const { topics = [], difficulty = 'MEDIUM', mode = 'INTERVIEW', timeLimit = null } = options;
        const isInterview = mode === 'INTERVIEW';
        set((state) => ({
            sessions: {
                ...state.sessions,
                [sessionId]: {
                    started: true,
                    paused: false,
                    mode,
                    score: isInterview ? 100 : null,
                    startTime: Date.now(),
                    elapsed: 0,
                    remaining: timeLimit,
                    timeLimit: timeLimit,
                    topics: isInterview ? topics : [],
                    difficulty: isInterview ? difficulty : null,
                },
            },
        }));
    },

    deductPoints: (sessionId, points) =>
        set((state) => {
            const session = state.sessions[sessionId];
            if (!session?.started || session.mode !== 'INTERVIEW') return state;
            return {
                sessions: {
                    ...state.sessions,
                    [sessionId]: { ...session, score: Math.max(0, session.score - points) },
                },
            };
        }),

    updateTimer: (sessionId, elapsed, remaining) =>
        set((state) => {
            const session = state.sessions[sessionId];
            if (!session || session.paused) return state;
            return {
                sessions: {
                    ...state.sessions,
                    [sessionId]: { ...session, elapsed, remaining },
                },
            };
        }),

    // New timer control methods
    pauseTimer: (sessionId) =>
        set((state) => {
            const session = state.sessions[sessionId];
            if (!session) return state;
            return {
                sessions: {
                    ...state.sessions,
                    [sessionId]: { ...session, paused: true },
                },
            };
        }),

    resumeTimer: (sessionId) =>
        set((state) => {
            const session = state.sessions[sessionId];
            if (!session) return state;
            // Adjust startTime so that elapsed remains correct after resume
            const now = Date.now();
            const newStartTime = now - (session.elapsed * 1000);
            return {
                sessions: {
                    ...state.sessions,
                    [sessionId]: { ...session, paused: false, startTime: newStartTime },
                },
            };
        }),

    resetTimer: (sessionId) =>
        set((state) => {
            const session = state.sessions[sessionId];
            if (!session) return state;
            return {
                sessions: {
                    ...state.sessions,
                    [sessionId]: {
                        ...session,
                        elapsed: 0,
                        remaining: session.timeLimit,
                        startTime: Date.now(),
                        paused: false,
                    },
                },
            };
        }),

    addTime: (sessionId, seconds) =>
        set((state) => {
            const session = state.sessions[sessionId];
            if (!session || !session.timeLimit) return state;
            const newTimeLimit = session.timeLimit + seconds;
            const newRemaining = (session.remaining || 0) + seconds;
            return {
                sessions: {
                    ...state.sessions,
                    [sessionId]: {
                        ...session,
                        timeLimit: newTimeLimit,
                        remaining: newRemaining,
                    },
                },
            };
        }),

    endSession: (sessionId) =>
        set((state) => {
            const session = state.sessions[sessionId];
            if (!session) return state;
            return {
                sessions: {
                    ...state.sessions,
                    [sessionId]: {
                        ...session,
                        started: false,
                        paused: true,
                        endedAt: Date.now(),
                        finalScore: session.mode === 'INTERVIEW' ? session.score : null,
                    },
                },
            };
        }),

    resetSession: (sessionId) =>
        set((state) => {
            const newSessions = { ...state.sessions };
            delete newSessions[sessionId];
            return { sessions: newSessions };
        }),

    getSession: (sessionId) => get().sessions[sessionId],

    getActiveInterviewSession: () => {
        const entries = Object.entries(get().sessions);
        const active = entries.find(([, session]) => session?.started && session.mode === 'INTERVIEW');
        return active ? { sessionId: active[0], ...active[1] } : null;
    },
}), {
    name: 'elitecode-interview-state',
    partialize: (state) => ({ sessions: state.sessions }),
}));

export default useInterviewStore;

const THEME_STORAGE_KEY = 'elitecode-theme'

  } catch {
    return DEFAULT_THEME
  }
}

  try {
    localStorage.setItem(THEME_STORAGE_KEY, theme)
  } catch {
    // ignore storage issues
  }
}

  theme: loadTheme(),
  setTheme: (theme) => {

    set({ theme })
  },
}))

import { create } from 'zustand'
import { persist } from 'zustand/middleware'
import useInterviewStore from './interviewStore'

const useChatStore = create(persist((set, get) => ({
  sessions: [],
  currentSessionId: null,
  currentSessionByMode: {},
  mode: 'STANDARD',
  isSidebarOpen: true,

  setMode: (mode) => {
    const modeSessionId = get().currentSessionByMode[mode] || null
    set({ mode, currentSessionId: modeSessionId })
  },
  toggleSidebar: () => set((s) => ({ isSidebarOpen: !s.isSidebarOpen })),

  createSession: (problemId = null, problemTitle = 'New Chat', modeOverride = null) => {
    const id = crypto.randomUUID();
    const sessionMode = modeOverride || get().mode;
    const session = {
      id,
      problemId,
      title: problemTitle,
      mode: sessionMode,
      messages: [],
      hasWork: false,
      createdAt: new Date().toISOString(),
    };
    set((s) => ({
      sessions: [session, ...s.sessions],
      currentSessionId: id,
      currentSessionByMode: { ...s.currentSessionByMode, [sessionMode]: id },
    }));
    return id;
  },

  setCurrentSession: (id) =>
      set((s) => {
        const session = s.sessions.find((item) => item.id === id)
        return {
          currentSessionId: id,
          currentSessionByMode: session
              ? { ...s.currentSessionByMode, [session.mode]: id }
              : s.currentSessionByMode,
        }
      }),

  addMessage: (sessionId, message, options = {}) =>
      set((s) => ({
        sessions: s.sessions.map((sess) =>
            sess.id === sessionId
                ? {
                    ...sess,
                    messages: [...sess.messages, message],
                    hasWork: options.countAsWork === false ? sess.hasWork : true,
                }
                : sess
        ),
      })),

  updateSessionTitle: (sessionId, title) =>
      set((s) => ({
        sessions: s.sessions.map((sess) =>
            sess.id === sessionId ? { ...sess, title } : sess
        ),
      })),

  markSessionWork: (sessionId) =>
      set((s) => ({
        sessions: s.sessions.map((sess) =>
            sess.id === sessionId ? { ...sess, hasWork: true } : sess
        ),
      })),

  deleteSession: (sessionId) =>
      set((s) => {
        // Also remove any interview store session data
        const { resetSession } = useInterviewStore.getState();
        resetSession(sessionId);

        const filtered = s.sessions.filter((sess) => sess.id !== sessionId)
        const deleted = s.sessions.find((sess) => sess.id === sessionId)
        const nextCurrentByMode = { ...s.currentSessionByMode }
        if (deleted && nextCurrentByMode[deleted.mode] === sessionId) {
          const replacement = filtered.find((sess) => sess.mode === deleted.mode)
          if (replacement) {
            nextCurrentByMode[deleted.mode] = replacement.id
          } else {
            delete nextCurrentByMode[deleted.mode]
          }
        }
        return {
          sessions: filtered,
          currentSessionId: filtered.find((sess) => sess.id === s.currentSessionId) ? s.currentSessionId : (nextCurrentByMode[s.mode] || null),
          currentSessionByMode: nextCurrentByMode,
        }
      }),

  getCurrentSession: () => {
    const { sessions, currentSessionId } = get()
    return sessions.find((s) => s.id === currentSessionId) || null
  },
}), {
  name: 'elitecode-chat-state',
  partialize: (state) => ({
    sessions: state.sessions,
    currentSessionId: state.currentSessionId,
    currentSessionByMode: state.currentSessionByMode,
    mode: state.mode,
    isSidebarOpen: state.isSidebarOpen,
  }),
}))

export default useChatStore

const useAuthStore = create((set) => ({
  user: null,
  accessToken: localStorage.getItem('accessToken') || null,
  isLoading: false,
  error: null,

    set({ user, accessToken, error: null });
  },
  clearAuth: () => {
    localStorage.removeItem('accessToken');
    set({ user: null, accessToken: null, error: null });
  },
  setLoading: (isLoading) => set({ isLoading }),
  setError: (error) => set({ error }),
  logout: () => {
    localStorage.removeItem('accessToken');
    set({ user: null, accessToken: null, error: null });
  },
}));

export default useAuthStore
